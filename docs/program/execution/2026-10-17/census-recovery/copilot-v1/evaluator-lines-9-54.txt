34:     def _ensure_asymmetric(self, graph: np.ndarray) -> np.ndarray:
35:         """
36:         Check if the graph is asymmetric and raise error if symmetry is found.
37:         
38:         Args:
39:             graph: Input adjacency matrix
40:             
41:         Raises:
42:             ValueError: If symmetric entries are found in the graph
43:             
44:         Returns:
45:             The input graph if it is asymmetric
46:         """
47:         n = graph.shape[0]
48:         
49:         for i in range(n):
50:             for j in range(i + 1, n):
51:                 if graph[i,j] == graph[j,i] and graph[i,j] != self.EDGE_TYPES['no_edge']:
52:                     raise ValueError(f"Symmetric entries found at positions ({i},{j}) and ({j},{i})")
53:         
54:         return graph
