290:     score_elements = function(y, d, g0_hat, g1_hat, m_hat, smpls) {
291:       if (is.character(self$score) && self$score == "ATTE") {
292:         # fraction of treated for ATTE
293:         p_hat = vector("numeric", length = self$data$n_obs)
294:         for (i_fold in seq_along(smpls$test_ids)) {
295:           p_hat[smpls$test_ids[[i_fold]]] = mean(
296:             self$data$data_model[[self$data$treat_col]][smpls$test_ids[[i_fold]]]
297:           )
298:         }
299:       }
300: 
301:       if (self$trimming_rule == "truncate" & self$trimming_threshold > 0) {
302:         m_hat[m_hat < self$trimming_threshold] = self$trimming_threshold
303:         m_hat[m_hat > 1 - self$trimming_threshold] = 1 - self$trimming_threshold
304:       }
305: 
306:       if (is.character(self$score)) {
307:         # compute residuals
308:         u0_hat = y - g0_hat
309:         if (self$score == "ATE") {
310:           u1_hat = y - g1_hat
311:           psi_b = g1_hat - g0_hat + d * (u1_hat) / m_hat -
312:             (1 - d) * u0_hat / (1 - m_hat)
313:           psi_a = rep(-1, self$data$n_obs)
314:         } else if (self$score == "ATTE") {
315:           psi_b = d * u0_hat / p_hat -
316:             m_hat * (1 - d) * u0_hat / (p_hat * (1 - m_hat))
317:           psi_a = -d / p_hat
318:         }
319:         psis = list(psi_a = psi_a, psi_b = psi_b)
320:       } else if (is.function(self$score)) {
321:         psis = self$score(
322:           y = y, d = d,
323:           g0_hat = g0_hat, g1_hat = g1_hat, m_hat = m_hat,
324:           smpls = smpls
325:         )
326:       }
327:       return(psis)
328:     },
