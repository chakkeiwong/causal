1: #' This calculates an upper and lower bound from bootstrap matrix
2: #'
3: #' This function takes a matrix containing the bootstrapped coefficients
4: #' from a parametric ADRF estimator and returns upper and lower 95 percent
5: #' confidence lines.
6: #'
7: #' @param grid_val is the vector of grid values on \code{treat} axis
8: #' @param coef_mat contains the bootstrapped parameter estimates.
9: #' @param degree is 1 for linear and 2 for quadratic outcome model
10: #'
11: #' @return
12: #' \code{get_ci} returns upper and lower 95 percent confidence lines.
13: #'
14: #' @usage
15: #'
16: #' get_ci(grid_val,
17: #'        coef_mat,
18: #'        degree)
19: #'
20: #' @export
21: #'
22: 
23: 
24: 
25: get_ci <- function(grid_val,
26:                    coef_mat,
27:                    degree){
28: 
29: 
30:   # gets the ci from bootstrapped parameter estimates
31: 
32:   if (degree == 1){
33: 
34: 
35:   grid_ci <- matrix( numeric(length(grid_val) * nrow(coef_mat) ), nrow = nrow(coef_mat))
36: 
37: 
38:   for ( i in 1:nrow(coef_mat)){
39:     grid_ci[i, ] <- coef_mat[i, 1] + coef_mat[i, 2] * grid_val
40:   }
41:   param_sorted <- apply(coef_mat, 2, sort, decreasing=F)
42: 
43:   # sort each column of grid_ci
44:   ci_sorted <- apply(grid_ci, 2, sort, decreasing=F)
45: 
46:   # get the 97.5% and 2.5% bands
47: 
48:   upper_ci <- ci_sorted[ceiling(0.975 * nrow(coef_mat)), ]
49:   lower_ci <- ci_sorted[floor(0.025 * nrow(coef_mat)), ]
50: 
51:   ci_est <- rbind(upper_ci, lower_ci)
52: 
53: 
54:   } else if (degree == 2){
55: 
56: 
57:     grid_ci <- matrix( numeric(length(grid_val) * nrow(coef_mat) ), nrow = nrow(coef_mat))
58: 
59: 
60:     for ( i in 1:nrow(coef_mat)){
61:       grid_ci[i, ] <- coef_mat[i, 1] + coef_mat[i, 2] * grid_val + coef_mat[i, 3] * grid_val^2
62:     }
63:     param_sorted <- apply(coef_mat, 2, sort, decreasing=F)
64: 
65:     # sort each column of grid_ci
66:     ci_sorted <- apply(grid_ci, 2, sort, decreasing=F)
67: 
68:     # get the 97.5% and 2.5% bands
69: 
70:     upper_ci <- ci_sorted[ceiling(0.975 * nrow(coef_mat)), ]
71:     lower_ci <- ci_sorted[floor(0.025 * nrow(coef_mat)), ]
72: 
73:     ci_est <- rbind(upper_ci, lower_ci)
74: 
75: 
76:   }
77:   return(list(ci_est))
78: 
79: }
