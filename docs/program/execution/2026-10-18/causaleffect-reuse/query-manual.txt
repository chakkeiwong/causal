1: \name{causal.effect}
2: \alias{causal.effect}
3: \title{Identify a causal effect}
4: \description{This function returns an expression for the joint distribution of the set of variables (\code{y})
5: given the intervention on the set of variables (\code{x}) conditional on (\code{z}) if the effect is identifiable.  Otherwise
6: an error is thrown describing the graphical structure that witnesses non-identifiability.  If \code{steps = TRUE}, returns instead
7: a list where the first element is the expression and the second element is a list of the intermediary steps taken by the algorithm.  }
8: 
9: \usage{causal.effect(y, x, z = NULL, G, expr = TRUE, simp = FALSE,
10:   steps = FALSE, primes = FALSE, prune = FALSE, stop_on_nonid = TRUE)}
11: 
12: \arguments{
13:   \item{y}{A character vector of variables of interest given the intervention. }
14:   \item{x}{A character vector of the variables that are acted upon. }
15:   \item{z}{A character vector of the conditioning variables. }
16:   \item{G}{An \code{igraph} object describing the directed acyclic graph induced by the causal model that matches the internal syntax. }
17:   \item{expr}{A logical value. If \code{TRUE}, a string is returned describing the expression in LaTeX syntax. Else, a list structure is returned which can be manually parsed by the function \code{get.expression}. }
18:   \item{simp}{A logical value. If \code{TRUE}, a simplification procedure is applied to the resulting probability object. d-separation and the rules of do-calculus are applied repeatedly to simplify the expression. }
19:   \item{steps}{A logical value. If \code{TRUE}, returns a list where the first element corresponds to the expression of the causal effect and the second to the a list describing intermediary steps taken by the algorithm.}
20:   \item{primes}{A logical value. If \code{TRUE}, prime symbols are appended to summation variables to make them distinct from their other instantiations. }
21:   \item{prune}{A logical value. If \code{TRUE}, additional steps are taken to remove variables that are not necessary for identification. }
22:   \item{stop_on_nonid}{A logical value. If \code{TRUE}, an error is produced when a non-identifiable effect is discovered. Otherwise recursion continues normally. }
23: }
24: 
25: \value{If \code{steps = FALSE}, A character string or an object of class \code{probability} that describes the interventional distribution. Otherwise, a list as described in the arguments.}
26: \references{
27: 
28: Shpitser I., Pearl J. 2006 Identification of Joint Interventional Distributions in Recursive semi-Markovian Causal Models.
29: \emph{Proceedings of the 21st National Conference on Artificial Intelligence}, \bold{2}, 1219--1226.
30: 
31: Shpitser I., Pearl J. 2006 Identification of Conditional Interventional Distributions.
32: \emph{Proceedings of the 22nd Conference on Uncertainty in Artificial Intelligence}, 427--444. }
33: 
34: \author{Santtu Tikka}
35: 
36: \seealso{ \code{\link{parse.graphml}}, \code{\link{get.expression}} }
37: 
38: \examples{
39: library(igraph)
40: 
41: # simplify = FALSE to allow multiple edges
42: g <- graph_from_literal(x -+ y, z -+ x, z -+ y , x -+ z, z -+ x, simplify = FALSE)
43: 
44: # Here the bidirected edge between X and Z is set to be unobserved in graph g
45: # This is denoted by giving them a description attribute with the value "U"
46: # The edges in question are the fourth and the fifth edge
47: g <- set_edge_attr(graph = g, name = "description", index = c(4, 5), value = "U")
48: causal.effect("y", "x", G = g)
49: 
50: # Pruning example
51: p <- graph_from_literal(
52:   x -+ z_4, z_4 -+ y, z_1 -+ x, z_2 -+ z_1,
53:   z_3 -+ z_2, z_3 -+ x, z_5 -+ z_1, z_5 -+ z_4, x -+ z_2, z_2 -+ x,
54:   z_3 -+ z_2, z_2 -+ z_3, z_2 -+ y, y -+ z_2,
55:   z_4 -+ y, y -+ z_4, z_5 -+ z_4, z_4 -+ z_5,
56:   simplify = FALSE
57: )
58: p <- set_edge_attr(p, "description", 9:18, "U")
59: causal.effect("y", "x", G = p, primes = TRUE, prune = TRUE)
60: 
61: # Simplification example
62: s <- graph_from_literal(x -+ y, w -+ x, w -+ z, z -+ y)
63: causal.effect("y", "x", G = s, simp = FALSE)
64: causal.effect("y", "x", G = s, simp = TRUE)
65: }
