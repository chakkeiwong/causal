#!/usr/bin/env python3
"""Independent finite arithmetic witnesses; no upstream code execution."""
from fractions import Fraction as F
import json
from pathlib import Path

def calculate():
    x=[F(-1),F(1)]; target=F(2); base=[F(0),F(1)]; lam=F(2)
    residual=target-sum(a*b for a,b in zip(x,base))
    adjustment=[residual*z/(sum(v*v for v in x)+lam) for z in x]
    weights=[a+b for a,b in zip(base,adjustment)]
    assert weights==[F(-1,4),F(5,4)] and sum(weights)==1
    outcomes=[F(80),F(120)];counterfactual=sum(a*b for a,b in zip(weights,outcomes))
    assert counterfactual==130 and F(150)-counterfactual==20
    full_errors={'A':[0,0,0,100],'B':[4,4,4,4]}
    selected={k:sum(v[:3])/F(3) for k,v in full_errors.items()}
    all_windows={k:sum(v)/F(4) for k,v in full_errors.items()}
    assert selected['A']<selected['B'] and all_windows['A']>all_windows['B']
    grid=[F(4)*F(1,4)**i for i in [-1,0,1]]
    assert grid==[16,4,1]
    perm=[F(0),F(1),F(2)];observed=F(3)
    p=F(sum(observed<=z for z in perm),len(perm))
    assert p==0 and F(1+sum(observed<=z for z in perm),len(perm)+1)==F(1,4)
    null_grid=[-1,0,1]+[0];assert null_grid.count(0)==2
    return {'provenance':'Project-authored exact arithmetic, not package output, empirical coverage or a borrowed theorem',
      'ridge':{'centered_donors':list(map(str,x)),'treated_centered':str(target),'lambda':str(lam),'simplex':list(map(str,base)),'adjustment':list(map(str,adjustment)),'weights':list(map(str,weights)),'sum':str(sum(weights)),'l1':str(sum(abs(v) for v in weights)),'pre_imbalance':str(target-sum(a*b for a,b in zip(x,weights))),'counterfactual_thousand_usd':str(counterfactual),'effect_thousand_usd':str(F(150)-counterfactual),'simplex_effect_thousand_usd':'30'},
      'holdout_witness':{'periods':4,'length':1,'inspected_starts':[1,2,3],'all_feasible_starts':[1,2,3,4],'abstract_losses':full_errors,'selected_means':{k:str(v) for k,v in selected.items()},'all_means':{k:str(v) for k,v in all_windows.items()},'scope':'Possible ranking change for abstract losses; not a realized package dataset'},
      'lambda_grid':{'nominal_max':'4','nominal_ratio':'1/16','n_lambda':2,'actual_values':list(map(str,grid)),'nominal_min':'1/4'},
      'inference':{'observed':'3','resampled':['0','1','2'],'raw_proportion':str(p),'plus_one_comparator':'1/4','zero_entries_after_append':null_grid.count(0),'scope':'Different finite conventions; no automatic calibration claim'},
      'economics':{'illustrative_usd':3600,'attempts':180,'completed':150,'scientifically_usable':120,'costs_usd':[str(F(3600,n)) for n in [180,150,120]]},
      'native_execution':False}

if __name__=='__main__':
    out=Path(__file__).resolve().parents[1]/'docs/program/execution/2026-10-24/augsynth-reuse-v1/exact-results.json'
    assert not out.exists();out.write_text(json.dumps(calculate(),indent=2)+'\n');print(json.dumps(calculate()))
