354:         )
355: 
356:         # compute residuals

358:         u_hat1 = y - g_hat1

361:             weights, weights_bar = self._get_weights(m_hat=m_hat_adj)
362:             psi_b = weights * (g_hat1 - g_hat0) + weights_bar * (
