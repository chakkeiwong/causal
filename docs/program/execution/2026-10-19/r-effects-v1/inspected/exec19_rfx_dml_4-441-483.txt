441:     #' Multiplier bootstrap for DoubleML models.
442:     #'
443:     #' @param method (`character(1)`) \cr
444:     #' A `character(1)` (`"Bayes"`, `"normal"` or `"wild"`) specifying the
445:     #' multiplier bootstrap method.
446:     #'
447:     #' @param n_rep_boot (`integer(1)`) \cr
448:     #' The number of bootstrap replications.
449:     #'
450:     #' @return self
451:     bootstrap = function(method = "normal", n_rep_boot = 500) {
452:       if (all(is.na(self$psi))) {
453:         stop("Apply fit() before bootstrap().")
454:       }
455:       assert_choice(method, c("normal", "Bayes", "wild"))
456:       assert_count(n_rep_boot, positive = TRUE)
457:       if (private$is_cluster_data) {
458:         stop("bootstrap not yet implemented with clustering.")
459:       }
460: 
461:       private$initialize_boot_arrays(n_rep_boot)
462: 
463:       for (i_rep in 1:self$n_rep) {
464:         private$i_rep = i_rep
465: 
466:         if (self$apply_cross_fitting) {
467:           n_obs = self$data$n_obs
468:         } else {
469:           smpls = private$get__smpls()
470:           test_ids = smpls$test_ids
471:           test_index = test_ids[[1]]
472:           n_obs = length(test_index)
473:         }
474:         weights = draw_weights(method, n_rep_boot, n_obs)
475: 
476:         for (i_treat in 1:self$data$n_treat) {
477:           private$i_treat = i_treat
478: 
479:           boot_res = private$compute_bootstrap(weights, n_rep_boot)
480:           i_start = (private$i_rep - 1) * private$n_rep_boot + 1
481:           i_end = private$i_rep * private$n_rep_boot
482:           private$boot_coef_[private$i_treat, i_start:i_end] = boot_res$boot_coef
483:           private$boot_t_stat_[private$i_treat, i_start:i_end] = boot_res$boot_t_stat
