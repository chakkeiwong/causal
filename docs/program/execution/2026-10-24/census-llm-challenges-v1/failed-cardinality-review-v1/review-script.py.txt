    reg=deepcopy(read(PRIOR/'census-register.json'));old=deepcopy(reg)
    row=next(x for x in reg['items'] if x['id']=='LA01-candidate-118');assert row['disposition']=='open_original_not_inspected'
    row.update(prior_disposition=row['disposition'],disposition='inspected_primary_retained_with_limits',technical_source_review_open=False,source_key=paper['integrated_citation_key'],primary_source_record=paper,publication_keys=[paper['integrated_citation_key'],project['integrated_citation_key']],task_families=['business_translation','diagram_construction','data_code','agent_architecture','evaluation_cost'],mapping_basis='Original survey taxonomy and limitations inform task-specific proposal controls; no new causal-discovery algorithm or native capability.',availability='Genuine72-page2023preprint acquired; selected text sections inspected; no visual/model/native execution.',effect_on_claims='Retain bounded proposal argument and exact authored counterexamples; withhold current rankings, behavioral fidelity and bank-value claims.',next_action='Complete underlying-original and task/version coverage, then preregister justified public/generated comparison with semantic/refusal oracles and full costs.')
