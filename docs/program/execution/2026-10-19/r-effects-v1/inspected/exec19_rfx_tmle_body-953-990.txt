954: #-------------------------------------------
955: .verifyArgs <- function(Y,Z,A,W,Delta, Qform, gform, g.Zform,g.Deltaform, obsWeights){
956: 	formulas <- list(Qform, gform, g.Zform, g.Deltaform)
957: 	validFormula <- sapply(formulas, function(x){identical(class(try(as.formula(x))), "formula")})
958: 	validNames <- c("Y", "Z", "A", ".", "Delta", colnames(W))
959: 	validTerms <- rep(TRUE, length(formulas))
960: 	validTerms[validFormula] <- sapply(formulas[which(validFormula)], 
961: 			function(x){is.null(x) || all(all.names(as.formula(x), functions=FALSE) %in% validNames)})
962: 	ok <- c(length(Y) == length(A) & length(A) == NROW(W) & length(A)==NROW(Delta),
963: 			sum(is.na(A), is.na(W)) == 0,
964: 			length(obsWeights) == length(Y),
965: 	 	 	all(A[!is.na(A)] %in% 0:1),
966: 	 	 	is.null(Z) || all(Z[!is.na(Z)] %in% 0:1),
967: 	 	 	is.null(Z) || length(unique(Z)) == 1 | (length(unique(Z)) > 1 & length(unique(A))>1),
968: 			validFormula,
969: 			validTerms
970: 			)
971: 	warning_messages <- c("\t'Y', 'A', 'W', 'Delta', must contain the same number of observations\n",
972: 				"\tNo missing values allowed in 'A' or 'W'\n",
973: 				"\tobsWeights must either be NULL or a vector of length n",
974: 				"\t'A' must be binary (0,1)\n",
975: 				"\t'Z' must be binary (0,1)\n",
976: 				"\tIntermediate variable (Z) not allowed when there is no experimentation in A",
977: 				"\tInvalid regression formula for 'Qform'",
978: 				"\tInvalid regression formula for 'gform'",
979: 				"\tInvalid regression formula for 'g.Zform'",
980: 				"\tInvalid regression formula for 'g.Deltaform'",
981: 				"\tInvalid term name in regression formula for 'Qform'",
982: 				"\tInvalid term name in regression formula for 'gform'",
983: 				"\tInvalid term name in regression formula for 'g.Zform'",
984: 				"\tInvalid term name in regression formula for 'g.Deltaform'"
985: 				)
986: 	if(!all(ok)){
987: 		warning("\n", warning_messages[!ok], immediate. = TRUE)
988: 	}
989: 	return(all(ok))
990: }
