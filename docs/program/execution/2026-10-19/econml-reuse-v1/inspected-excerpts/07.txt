937:         if self.transformer:

945:         if self.discrete_outcome:
946:             Y = self.outcome_transformer.transform(Y).reshape(-1, 1)

949:             folds = None

953:             if splitter != self.cv and isinstance(splitter, (KFold, StratifiedKFold)):

964:             to_split = np.hstack(all_vars) if all_vars else np.ones((T.shape[0], 1))
965:

971:                                     "Instead you must initialize this object with a splitter that can handle groups.")

977:                                                                   self.use_ray, self.ray_remote_func_options, Y, T,

979:                                                                   groups=groups)
