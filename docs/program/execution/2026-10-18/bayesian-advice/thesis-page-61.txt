 5   dag <- empty_dag()

 7   dag <- dag +
 8     node("summer", type="rnorm", mean=0, sd=1) +
 9     node("sun", type="gaussian", parents=c("summer"), betas=c(runif(n=1, min
           =50, max=100)), intercept=runif(n=1, min=1, max=10), error=runif(n=1,
           min=0, max=5)) +
10     node("vacation", type="gaussian", parents=c("summer", "sun"), betas=runif(n
           =2, min=50, max=100), intercept=runif(n=1, min=1, max=10), error=runif(
           n=1, min=0, max=5)) +
11     node("beach", type="gaussian", parents=c("sun", "vacation"), betas=runif(n
           =2, min=50, max=100), intercept=runif(n=1, min=1, max=10), error=runif(
           n=1, min=0, max=5)) +
12     node("sunburn", type="gaussian", parents=c("sun"), betas=c(runif(n=1, min
           =50, max=100)), intercept=runif(n=1, min=1, max=10), error=runif(n=1,
           min=0, max=5)) +
13     node("drowning", type="gaussian", parents=c("beach"), betas=c(runif(n=1,
           min=50, max=100)), intercept=runif(n=1, min=1, max=10), error=runif(n=1,
            min=0, max=5)) +
14     node("hospital", type="gaussian", parents=c("sunburn", "drowning"), betas=
           runif(n=2, min=50, max=100), intercept=runif(n=1, min=1, max=10), error
           =runif(n=1, min=0, max=5))
