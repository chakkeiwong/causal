def prepare():

        for name, specs in SPECS.items():

            assert tar.extractfile(saved['archive_member']).read()==data
            assert core.sha(p)==saved['local_copy']['sha256']
            lines=data.decode().splitlines(); anchors=[]
            for lo,hi,finding in specs:
                assert 1<=lo<=hi<=len(lines),(name,hi,len(lines))
                anchors.append(dict(lines=[lo,hi],finding=finding,
                    excerpt_sha256=hashlib.sha256(chr(10).join(lines[lo-1:hi]).encode()).hexdigest()))
            records.append(dict(source_identity='https://github.com/cran/fect/blob/'+PIN+'/'+name,
                upstream_path=name,version_date='2.4.5;30 May 2026;'+PIN,

def receipt():
    state=core.read(ROOT,core.STATE)
    assert state['execution_runtime']['active']['operation']==OP
    verify_bindings(read(BASE/'review.json'))
    graph,merged,_=run.combined(ROOT,state); node=run.ledger.operation(graph,OP)
    value=read(BASE/'choice.json');value.update(status='pass',scope='Completed proposal reference decision;no native admission',evidence=[ref(BASE/'review.json','Author review')])
    output=ROOT/node['outputs'][0];write(output,value)
    evidence=[ref(BASE/n,'Reviewed decision') for n in ['review.json','choice.json','dependency-license-review.json','conversion-contract.json','exact-results.json']]
    findings=['Pinned archive identity,license declaration,dependencies and maintenance limits explicit.',
        'Targets,clocks,numerical/inference contracts,alternatives,owners,costs and revisit trigger reviewed.',
        'Reference only;integration false and capabilities empty;native public qualification remains conditional.']
    result=run.ledger.receipt_template(node)
    result.update(outcome='pass',execution='performed',scope=value['scope'],reviewer=dict(name='Codex author/coordinator',kind='author'),
        outputs=[ref(output,'Reference disposition')],criteria=[dict(index=i,verdict='pass',finding=f,evidence=evidence) for i,f in enumerate(findings)],
        limitations=LIMITS,capabilities=[],nonexecution_reason='',revisit_trigger=value['revisit_trigger'])
    run.validate_receipt(ROOT,merged,node,result,len(state['execution_program']['turns']));verify_bindings(result)
    write(BASE/'operation-receipt.json',result);print('fect reference receipt prepared.')
